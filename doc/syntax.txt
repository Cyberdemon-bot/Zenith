(* ============ TOP LEVEL ============ *)
program          := statement* EOF

(* ============ STATEMENTS ============ *)
statement        := ";"* ( let_stmt
                         | return_stmt
                         | break_stmt
                         | continue_stmt
                         | assign_stmt
                         | expr_stmt
                         )

let_stmt         := "let" typed_identifier "=" expr ";"
typed_identifier := IDENT ( ":" type_annotation )?
type_annotation  := "&"? "mut"? IDENT

return_stmt      := "return" expr? ";"
break_stmt       := "break" LABEL? expr? ";"
continue_stmt    := "continue" ";"

assign_stmt      := assign_target "=" expr ";"
assign_target    := IDENT | field_access | index_expr

expr_stmt        := stmt_expr ";"?
stmt_expr        := block_expr
                  | labeled_block_expr
                  | if_expr
                  | call_expr

(* ============ EXPRESSIONS ============ *)
expr             := prefix_expr (infix_expr)*     (* precedence-climbing *)

prefix_expr      := INTEGER
                  | DECIMAL
                  | STRING
                  | "true" | "false" | "null"
                  | "!"  expr
                  | "-"  expr
                  | "&"  expr
                  | "("  expr ")"
                  | IDENT
                  | block_expr
                  | labeled_block_expr
                  | if_expr

infix_expr       := binop expr
                  | "." IDENT                     (* field access *)
                  | "(" arg_list ")"              (* call *)
                  | "[" expr "]"                  (* index *)

binop            := "+" | "-" | "*" | "/" | "%" | "**"
                  | "==" | "!=" | "<" | ">" | "<=" | ">="

arg_list         := (expr ("," expr)* ","?)?

block_expr       := "{" statement* "}"
labeled_block_expr := LABEL ":" block_expr

if_expr          := "if" expr then_clause else_clause?
then_clause      := "=>" expr | block_expr
else_clause      := "else" ( if_expr | "=>" expr | block_expr )

(* ============ PRECEDENCE (low → high) ============ *)
P_LOWEST       = 0
P_EQUALS       = 1     (* == != *)
P_LESSGREATER  = 2     (* < > <= >= *)
P_SUM          = 3     (* + - *)
P_PRODUCT      = 4     (* * / % *)
P_UNARY        = 5     (* ! - & *)
P_EXPONENT     = 6     (* ** — right-assoc *)
P_CALL         = 7     (* . và ( *)
P_INDEX        = 8     (* [ *)
